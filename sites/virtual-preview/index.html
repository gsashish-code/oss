What's inside

Everything you need. Nothing you don't.

VirtualList

Static or dynamic row heights. ResizeObserver keeps sizes accurate as content changes.

VirtualGrid

Row and column virtualization. Only the cells in view are mounted — no wasted renders.

Chat scroll

Anchored-to-bottom mode. Prepend history without jumping. Stick-to-bottom when new messages arrive.

Framework-free core

The Virtualizer class has zero dependencies. The React adapter is an optional layer on top.

As simple as it gets

import { VirtualList } from '@gsashish/virtual/react'

const rows = Array.from({ length: 100_000 }, (_, i) => `Row ${i}`)

export function App() {
  return (
    <VirtualList
      count={rows.length}
      estimateSize={() => 48}
      height={600}
      renderItem={(index) => (
        <div className="row">{rows[index]}</div>
      )}
    />
  )
}
import { useRef } from 'react'
import { useVirtualizer } from '@gsashish/virtual/react'

export function List({ items }) {
  const ref = useRef(null)
  const { items: rows, totalSize, measureRef } = useVirtualizer({
    count: items.length,
    estimateSize: (i) => items[i].height ?? 48,
    getScrollElement: () => ref.current,
  })

  return (
    <div ref={ref} style={{ height: 600, overflow: 'auto' }}>
      <div style={{ height: totalSize, position: 'relative' }}>
        {rows.map((row) => (
          <div key={row.key} ref={measureRef(row.index)}
            style={{ position: 'absolute', top: 0, width: '100%',
              transform: `translateY(${row.start}px)` }}>
            {items[row.index].text}
          </div>
        ))}
      </div>
    </div>
  )
}
import { useChatVirtualizer } from '@gsashish/virtual/react'

export function ChatWindow({ messages, loadMore }) {
  const { items, totalSize, scrollRef, measureRef,
          scrollToBottom, isStuckToBottom } = useChatVirtualizer({
    count: messages.length,
    estimateSize: () => 56,
    onReachedTop: loadMore,  // infinite scroll upward
  })

  return (
    <div ref={scrollRef} style={{ height: 600, overflow: 'auto' }}>
      <div style={{ height: totalSize, position: 'relative' }}>
        {items.map((item) => (
          <div key={item.key} ref={measureRef(item.index)}
            style={{ position: 'absolute', top: 0, width: '100%',
              transform: `translateY(${item.start}px)` }}>
            {messages[item.index].text}
          </div>
        ))}
      </div>
      {!isStuckToBottom &&
        <button onClick={scrollToBottom}>↓ New messages</button>}
    </div>
  )
}
